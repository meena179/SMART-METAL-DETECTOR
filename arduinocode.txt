// Impact detection
const float IMPACT_THRESHOLD_G = 4.0;
const unsigned long COUNTDOWN_MS = 30000;
const unsigned long ALERT_SCREEN_MS = 5000;
const unsigned long GPS_MAX_AGE_MS = 120000;

enum SystemState {
  MONITORING,
  CONFIRMING,
  ALERT_DISPLAY
};

SystemState state = MONITORING;

unsigned long countdownStart = 0;
unsigned long lastScreenUpdate = 0;
unsigned long lastBeepToggle = 0;
unsigned long alertScreenStart = 0;

unsigned long belowThresholdStart = 0;
bool belowThresholdTiming = false;
bool armed = true;

bool buzzerOn = false;
bool lastButtonReading = HIGH;
bool stableButtonState = HIGH;
unsigned long lastButtonChange = 0;

float totalG = 1.0;

// ---------- GPS ----------

void readGPS() {
  while (GPSserial.available()) {
    gps.encode(GPSserial.read());
  }
}

bool gpsLocationAvailable() {
  return gps.location.isValid() &&
         gps.location.age() < GPS_MAX_AGE_MS;
}

// ---------- MPU6050 ----------

void writeMPU(byte reg, byte value) {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(reg);
  Wire.write(value);
  Wire.endTransmission();
}

bool readAcceleration() {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x3B);

  if (Wire.endTransmission(false) != 0) {
    return false;
  }

  if (Wire.requestFrom(MPU_ADDR, 6, true) != 6) {
    return false;
  }

  int16_t ax = (int16_t)((Wire.read() << 8) | Wire.read());
  int16_t ay = (int16_t)((Wire.read() << 8) | Wire.read());
  int16_t az = (int16_t)((Wire.read() << 8) | Wire.read());

  // +/-16g range: 2048 raw counts per g
  float xg = ax / 2048.0;
  float yg = ay / 2048.0;
  float zg = az / 2048.0;

  totalG = sqrt(xg*xg + yg*yg + zg*zg);
  return true;
}

// ---------- Button debounce ----------

bool buttonPressed() {
  bool reading = digitalRead(BUTTON_PIN);

  if (reading != lastButtonReading) {
    lastButtonChange = millis();
    lastButtonReading = reading;
  }

  if (millis() - lastButtonChange >= 35) {
    if (reading != stableButtonState) {
      stableButtonState = reading;
      if (stableButtonState == LOW) {
        return true;
      }
    }
  }

  return false;
}

// ---------- Wi-Fi ----------

bool connectWiFi() {
  if (WiFi.status() == WL_CONNECTED) {
    return true;
  }

  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  Serial.print("Connecting to Wi-Fi");
  unsigned long start = millis();

  while (WiFi.status() != WL_CONNECTED &&
         millis() - start < 20000) {
    readGPS();
    delay(250);
    Serial.print(".");
  }

  Serial.println();

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("Wi-Fi connected");
    Serial.println(WiFi.localIP());
    return true;
  }

  Serial.println("Wi-Fi connection failed");
  return false;
}

// ---------- Telegram accident message ----------

bool sendTelegramAlert() {
  // Process any GPS data already waiting in the UART buffer.
  readGPS();

  bool hasGPS = gpsLocationAvailable();

  if (!connectWiFi()) {
    Serial.println("Cannot send: Wi-Fi unavailable");
    return false;
  }

  // Process GPS again after Wi-Fi connection.
  readGPS();
  hasGPS = gpsLocationAvailable();

  String message =
    "ACCIDENT ALERT\n\n"
    "An impact was detected by the ESP32 system.\n"
    "The 30-second countdown ended without cancellation.\n\n";

  if (hasGPS) {
    double latitude = gps.location.lat();
    double longitude = gps.location.lng();

    message += "GPS LOCATION:\n";
    message += "Latitude: " + String(latitude, 6) + "\n";
    message += "Longitude: " + String(longitude, 6) + "\n";
    message += "Google Maps: https://maps.google.com/?q=";
    message += String(latitude, 6) + ",";
    message += String(longitude, 6);
  } else {
    message +=
      "GPS LOCATION NOT AVAILABLE.\n"
      "The GPS has not provided a recent valid fix.\n"
      "Please verify the location manually.";
  }

  Serial.println("Sending Telegram accident alert...");

  bool sent = bot.sendMessage(CHAT_ID, message, "");

  Serial.println(sent
    ? "Telegram alert sent successfully"
    : "Telegram alert failed");

  Serial.println(hasGPS
    ? "GPS fix included in message"
    : "No valid GPS fix; unavailable notice included");

  return sent;
}

// ---------- Setup ----------

void setup() {
  Serial.begin(115200);

  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);

  digitalWrite(BUZZER_PIN, LOW);
  digitalWrite(LED_PIN, LOW);

  Wire.begin(SDA_PIN, SCL_PIN);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("OLED not found at address 0x3C");
    while (true) delay(1000);
  }

  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("Starting system...");
  display.display();

  // Start GPS serial communication
  GPSserial.begin(9600, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);

  // Wake MPU6050
  writeMPU(0x6B, 0x00);
  delay(100);

  // Set accelerometer to +/-16g
  writeMPU(0x1C, 0x18);
  delay(100);

  // Prototype only: skips TLS certificate verification.
  // Use proper certificate verification for deployment.
  secureClient.setInsecure();

  Serial.println("Accident Guard starting");
  Serial.println("GPS searching for a fix...");

  connectWiFi();
  showReady();
}

// ---------- Main loop ----------

void loop() {
  readGPS();

  unsigned long now = millis();
  bool pressed = buttonPressed();

  if (!readAcceleration()) {
    Serial.println("MPU6050 read failed");
    delay(100);
    return;
  }

  // Re-arm only after acceleration remains below 1.5g
  // for 2 seconds following an event.
  if (!armed && state == MONITORING) {
    if (totalG < 1.5) {
      if (!belowThresholdTiming) {
        belowThresholdStart = now;
        belowThresholdTiming = true;
      }

      if (now - belowThresholdStart >= 2000) {
        armed = true;
        belowThresholdTiming = false;
        Serial.println("System re-armed");
      }
    } else {
      belowThresholdTiming = false;
    }
  }

  // Cancel alert
  if (state == CONFIRMING && pressed) {
    state = MONITORING;
    armed = false;
    belowThresholdTiming = false;

    digitalWrite(BUZZER_PIN, LOW);
    digitalWrite(LED_PIN, LOW);
    buzzerOn = false;

    Serial.println("Alert cancelled");
    showCancelled();
    delay(1200);
    showReady();
    return;
  }

  // Monitor impact
  if (state == MONITORING) {
    digitalWrite(LED_PIN, LOW);

    if (armed && totalG >= IMPACT_THRESHOLD_G) {
      state = CONFIRMING;
      armed = false;
      belowThresholdTiming = false;

      countdownStart = now;
      lastScreenUpdate = 0;
      lastBeepToggle = now;
      buzzerOn = true;

      digitalWrite(BUZZER_PIN, HIGH);
      digitalWrite(LED_PIN, HIGH);

      Serial.print("Impact detected: ");
      Serial.print(totalG);
      Serial.println(" g");

      showImpact();
    }
  }

  // 30-second countdown
  else if (state == CONFIRMING) {
    unsigned long elapsed = now - countdownStart;

    if (now - lastBeepToggle >= 250) {
      lastBeepToggle = now;
      buzzerOn = !buzzerOn;
      digitalWrite(BUZZER_PIN, buzzerOn ? HIGH : LOW);
    }

    digitalWrite(
      LED_PIN, ((now / 250) % 2 == 0) ? HIGH : LOW
    );

    if (now - lastScreenUpdate >= 200) {
      lastScreenUpdate = now;

      unsigned long remaining =
        elapsed >= COUNTDOWN_MS ? 0 : COUNTDOWN_MS - elapsed;

      showCountdown(remaining);
    }

    if (elapsed >= COUNTDOWN_MS) {
      digitalWrite(BUZZER_PIN, LOW);
      digitalWrite(LED_PIN, LOW);
      buzzerOn = false;

      state = ALERT_DISPLAY;

      // Capture whether a fresh GPS fix is available.
      bool hasGPS = gpsLocationAvailable();
      bool sent = sendTelegramAlert();

      // Refresh GPS status after sending.
      hasGPS = gpsLocationAvailable();
      showAlertSent(sent, hasGPS);

      alertScreenStart = millis();
    }
  }

  // Show result, then resume monitoring disarmed
  else if (state == ALERT_DISPLAY) {
    if (now - alertScreenStart >= ALERT_SCREEN_MS) {
      state = MONITORING;
      belowThresholdTiming = false;
      showReady();
    }
  }

  delay(10);
}
